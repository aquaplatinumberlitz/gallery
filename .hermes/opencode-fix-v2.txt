ISSUE 4: Test reactivate_existing=False
====================================

File mới: backend/tests/test_asset_reactivate.py

Test rằng metadata upsert KHÔNG resurrect offline/deleted asset.

```python
"""Test that metadata operations do not resurrect offline/deleted assets."""

import time
from pathlib import Path

import pytest


def test_metadata_upsert_does_not_reactivate_offline_asset(
    metadata_db, temp_assets_root, library_fixture
):
    """upsert_extracted_metadata should not resurrect an offline/deleted asset."""
    from backend.metadata_store import index_file, upsert_extracted_metadata
    from backend.metadata_store._db import _connect
    from backend.metadata_extract import ExtractedMetadata

    lib = library_fixture
    asset_path = temp_assets_root / "test_offline.png"
    asset_path.write_text("fake png")

    # Index it
    index_file(str(asset_path))

    # Mark as offline + deleted (simulate reconcile)
    conn = _connect()
    conn.execute(
        "UPDATE assets SET offline = 1, deleted_at = ? WHERE path = ?",
        (time.time(), str(asset_path)),
    )
    conn.commit()

    # Now upsert metadata — this should NOT resurrect
    meta = ExtractedMetadata(path=str(asset_path), name="test_offline.png")
    upsert_extracted_metadata(meta)

    # Verify still offline
    row = conn.execute(
        "SELECT offline, deleted_at FROM assets WHERE path = ?",
        (str(asset_path),),
    ).fetchone()
    assert row is not None
    assert row["offline"] == 1, "Asset should remain offline after metadata upsert"
    assert row["deleted_at"] is not None, (
        "Asset should remain deleted after metadata upsert"
    )


def test_index_file_reactivates_active_asset(
    metadata_db, temp_assets_root, library_fixture
):
    """Normal index operation (reactivate_existing=True default) should reactivate."""
    from backend.metadata_store import index_file
    from backend.metadata_store._db import _connect

    lib = library_fixture
    asset_path = temp_assets_root / "test_active.png"
    asset_path.write_text("fake png")

    index_file(str(asset_path))

    # Verify it's active
    conn = _connect()
    row = conn.execute(
        "SELECT offline, deleted_at FROM assets WHERE path = ?",
        (str(asset_path),),
    ).fetchone()
    assert row["offline"] == 0
    assert row["deleted_at"] is None
```
